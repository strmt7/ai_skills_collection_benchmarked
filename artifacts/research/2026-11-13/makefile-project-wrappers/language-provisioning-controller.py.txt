"""Run a frozen offline component with optional pinned SDK and small assets."""
import argparse
import ctypes
import hashlib
import json
import os
import re
import sys
from pathlib import Path

S = Path(__file__).resolve().parent
R = S.parents[2]
sys.path.insert(0, str(R / 'tools'))
from _native_benchmark_runtime import ContainerLease, DockerClient, LeaseResources, stage_inputs
from _windows_trial_process import run_owned


def owned_file(path):
    path = path.resolve(strict=True)
    if not path.is_relative_to(S) or not path.is_file():
        raise ValueError('Declared component inputs must be owned regular files')
    return path


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--program', type=Path, required=True)
    parser.add_argument('--output', type=Path, required=True)
    parser.add_argument('--sdk-capture', type=Path)
    parser.add_argument('--assets', type=Path)
    args = parser.parse_args()
    program = owned_file(args.program)
    out = args.output.resolve()
    assert out.is_relative_to(S) and out != S and not out.exists()
    raw = program.read_bytes()
    assert len(raw) <= 64 * 1024
    files = {'probe.py': raw}
    if args.assets:
        for name, item in json.loads(owned_file(args.assets).read_bytes()).items():
            assert re.fullmatch(r'[A-Za-z0-9_.+-]+(?:/[A-Za-z0-9_.+-]+)*', name) and '..' not in name.split('/') and name not in files
            path = owned_file(Path(item['path']))
            data = path.read_bytes()
            assert len(data) == item['bytes'] and hashlib.sha256(data).hexdigest() == item['sha256']
            files[name] = data
    assert len(files) <= 128 and sum(map(len, files.values())) <= 63 * 1024**2
    sdk = capture = None
    if args.sdk_capture:
        capture_path = owned_file(args.sdk_capture)
        capture = json.loads(capture_path.read_bytes())
        assert Path(capture['archive_name']).name == capture['archive_name']
        sdk = owned_file(capture_path.parent / capture['archive_name'])
        assert sdk.stat().st_size == capture['bytes'] <= 256 * 1024**2 and re.fullmatch(r'[0-9a-f]{64}', capture['sha256'])
        assert hashlib.sha256(sdk.read_bytes()).hexdigest() == capture['sha256']
    class MemoryStatus(ctypes.Structure):
        _fields_ = [('length', ctypes.c_ulong), ('load', ctypes.c_ulong)] + [(name, ctypes.c_ulonglong) for name in ('total', 'available', 'page_total', 'page_available', 'virtual_total', 'virtual_available', 'extended_available')]
    memory = MemoryStatus()
    memory.length = ctypes.sizeof(memory)
    if not ctypes.windll.kernel32.GlobalMemoryStatusEx(ctypes.byref(memory)):
        raise ctypes.WinError()
    assert memory.available >= 8 * 1024**3
    out.mkdir()
    runtime_path = S / 'native-controller-runtime-qualification-v1/runtime.json'
    runtime = json.loads(runtime_path.read_bytes())
    exe = Path(runtime['docker_cli'])
    assert hashlib.sha256(exe.read_bytes()).hexdigest() == runtime['docker_cli_sha256']
    config = out / 'preflight'
    config.mkdir()
    (config / 'config.json').write_text('{"auths":{}}\n', encoding='utf-8')
    env = {key: os.environ[key] for key in ('SYSTEMROOT', 'WINDIR', 'TEMP', 'TMP') if key in os.environ}
    env['PATH'] = os.pathsep.join((str(exe.parent), str(Path(os.environ['SYSTEMROOT']) / 'System32')))
    with open(os.devnull, 'rb') as stdin, (out / 'preflight.stdout.native').open('xb') as stdout, (out / 'preflight.stderr.native').open('xb') as stderr:
        observed = run_owned([str(exe), '--host', runtime['endpoint'], '--config', str(config), 'image', 'inspect', runtime['image_reference']], cwd=R, environment=env, stdin=stdin, stdout=stdout, stderr=stderr, timeout=20, output_limit=1024**2)
    assert observed['exit_code'] == 0 and observed['job_active_processes_after_cleanup'] == 0, 'No uncached image pull admitted'
    client = DockerClient(R, out / 'client', runtime_path)
    lease = ContainerLease(client, out / 'lease.json', seconds=300, resources=LeaseResources(memory_bytes=2 * 1024**3, tmpfs_bytes=1024**3, executable_workspace=True))
    try:
        lease.create()
        lease.start()
        if sdk:
            assert lease.identity is not None and lease.owned(lease.identity) is not None
            provisioning = out / 'sdk-provision'
            provisioning.mkdir()
            receiver = """import hashlib,json,os,sys
from pathlib import Path
size=int(sys.argv[1]);expected=sys.argv[2]
if not 0<size<=256*1024**2:raise ValueError('Invalid SDK size')
digest=hashlib.sha256();total=0;path=Path('/tmp/qualification-sdk.tar.gz')
with path.open('xb') as target:
 while block:=sys.stdin.buffer.read(1024**2):
  total+=len(block)
  if total>size:raise ValueError('SDK exceeds declared size')
  digest.update(block);target.write(block)
if total!=size or digest.hexdigest()!=expected:raise ValueError('SDK size/digest mismatch')
path.chmod(0o444)
print(json.dumps({'bytes':total,'sha256':digest.hexdigest(),'uid':os.getuid()}))
"""
            argv = [str(client.docker), '--host', client.endpoint, '--config', str(client.config), 'container', 'exec', '--interactive', '--user=dev', lease.identity, 'python', '-I', '-B', '-c', receiver, str(capture['bytes']), capture['sha256']]
            with sdk.open('rb') as stdin, (provisioning / 'stdout.native').open('xb') as stdout, (provisioning / 'stderr.native').open('xb') as stderr:
                process = run_owned(argv, cwd=R, environment=client.environment, stdin=stdin, stdout=stdout, stderr=stderr, timeout=60, output_limit=1024**2, memory_limit=512*1024**2, process_limit=16)
            (provisioning / 'process.json').write_text(json.dumps({'argv': argv, 'process': process}, indent=2) + '\n', encoding='utf-8')
            assert process['exit_code'] == 0 and process['termination_reason'] is None and process['job_active_processes_after_cleanup'] == 0
            ack = json.loads((provisioning / 'stdout.native').read_bytes())
            assert ack['bytes'] == capture['bytes'] and ack['sha256'] == capture['sha256'] and ack['uid'] > 0
        stage_inputs(lease, out / 'staging', files)
        observation = json.loads(client.run('language-component', ['container', 'exec', '--user=dev', lease.identity, 'python', '-I', '-B', '/tmp/benchmark-inputs/probe.py'], timeout=240))
    finally:
        lease.close()
    assert not client.ids(lease.token) and program.read_bytes() == raw
    summary = {'observation': observation, 'program_sha256': hashlib.sha256(raw).hexdigest(), 'asset_sha256': {name: hashlib.sha256(data).hexdigest() for name, data in files.items()}, 'sdk_sha256': capture['sha256'] if capture else None, 'image_reference': runtime['image_reference'], 'available_host_ram_before_bytes': memory.available, 'memory_limit_bytes': 2*1024**3, 'tmpfs_limit_bytes': 1024**3, 'lease_seconds': 300, 'lease_closed': True, 'remaining_owned_containers': 0, 'images_pulled': 0, 'model_invocations': 0, 'global_installation': False}
    (out / 'summary.json').write_text(json.dumps(summary, indent=2) + '\n', encoding='utf-8', newline='\n')
    print(json.dumps({'passed': observation.get('passed'), 'cases': len(observation.get('cases', [])), 'remaining_owned_containers': 0}))


if __name__ == '__main__':
    main()
