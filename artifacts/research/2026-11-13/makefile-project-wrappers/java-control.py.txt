def write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding='utf-8', newline='\n')


def run(argv, cwd, env, timeout=30):
    result = subprocess.run(argv, cwd=cwd, env=env, stdin=subprocess.DEVNULL, capture_output=True, text=True, timeout=timeout)
    row = {'argv': argv, 'cwd_name': cwd.name, 'exit_code': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr}
    records.append(row)
    print(json.dumps({'native_command': row}), file=sys.stderr, flush=True)
    return result

    release = {}
    for line in (sdk / 'release').read_text().splitlines():
        key, sep, value = line.partition('=')
        if sep and key in {'JAVA_VERSION', 'IMPLEMENTOR', 'IMPLEMENTOR_VERSION', 'OS_ARCH', 'OS_NAME'}:
            release[key] = value.strip('"')
    versions = {}
    for name, executable, args in [('java', java, ['--version']), ('javac', javac, ['--version']), ('jar', jar, ['--version'])]:
        result = run([executable, *args], root, env)
        assert result.returncode == 0 and release['JAVA_VERSION'] in result.stdout + result.stderr
        versions[name] = {'stdout': result.stdout, 'stderr': result.stderr}
    case('actual-sdk-tool-identities', release=release)
    junit = Path('/tmp/benchmark-inputs/junit.jar')
    assert hashlib.sha256(junit.read_bytes()).hexdigest() == CONTRACT['junit']['sha256']
    help_result = run([java, '-jar', str(junit), 'execute', '--help'], root, env)
    assert help_result.returncode == 0 and '--fail-if-no-tests' in help_result.stdout and '--scan-class-path' in help_result.stdout
    case('actual-console-feature-admission', version=CONTRACT['junit']['version'])
    generator = root / 'generate.sh'
    write(generator, GENERATOR)

    def generated(project):
        project.mkdir()
        result = run([bash, str(generator), 'java', 'demo', 'Makefile'], project, env)
        assert result.returncode == 0

    def build(project, target):
        return run([make, '--no-print-directory', target], project, env)

    removal = root / 'removal'
    generated(removal)
    write(removal / 'src/Main.java', 'public class Main { public static void main(String[] args) { System.out.print("live"); } }\n')
    write(removal / 'src/Obsolete.java', 'public class Obsolete { }\n')
    assert build(removal, 'build').returncode == 0
    original_jar = removal / 'dist/demo-1.0.0.jar'
    assert run([java, '-jar', str(original_jar)], removal, env).stdout == 'live'
    with zipfile.ZipFile(original_jar) as source:
        assert 'Obsolete.class' in source.namelist()
    case('original-initial-archive', executable=True)
    (removal / 'src/Obsolete.java').unlink()
    assert build(removal, 'build').returncode == 0
    with zipfile.ZipFile(original_jar) as source:
        assert 'Obsolete.class' in source.namelist()
    case('original-removed-source-retains-class', obsolete_class_retained=True)
    fresh = removal / 'fresh-classes'
    fresh.mkdir()
    assert run([javac, '-d', str(fresh), str(removal / 'src/Main.java')], removal, env).returncode == 0
    fresh_jar = removal / 'fresh.jar'
    assert run([jar, '--create', '--file', str(fresh_jar), '--main-class', 'Main', '-C', str(fresh), '.'], removal, env).returncode == 0
    with zipfile.ZipFile(fresh_jar) as source:
        assert 'Main.class' in source.namelist() and 'Obsolete.class' not in source.namelist()
    case('fresh-native-output-excludes-removed-source', obsolete_class_retained=False)
    missing_tests = build(removal, 'test')
    assert missing_tests.returncode == 0 and 'No test directory found' in missing_tests.stdout
    case('original-no-test-directory', exit_code=0, tests_run=False)

    dependency = root / 'dependency'
    generated(dependency)
    write(dependency / 'src/Main.java', 'public class Main { public static void main(String[] args) { System.out.print(dep.Helper.value()); } }\n')
    (dependency / 'lib').mkdir()
    dep_jar = dependency / 'lib/dependency.jar'
    for index, code in enumerate(['package dep; public class Helper { public static int value() { return 42; } }', 'package dep; public class Helper { public static int changed() { return 43; } }']):
        source = root / f'library-{index}/dep/Helper.java'
        write(source, code + '\n')
        classes = root / f'library-{index}/classes'
        classes.mkdir()
        assert run([javac, '-d', str(classes), str(source)], root, env).returncode == 0
        assert run([jar, '--create', '--file', str(dep_jar), '-C', str(classes), '.'], root, env).returncode == 0
        built = build(dependency, 'build')
        assert built.returncode == 0
        if index == 0:
            assert run([java, '-cp', 'build/classes:lib/*', 'Main'], dependency, env).stdout == '42'
    case('original-changed-abi-build-reports-success', exit_code=0)
    runtime = run([java, '-cp', 'build/classes:lib/*', 'Main'], dependency, env)
    assert runtime.returncode != 0 and 'NoSuchMethodError' in runtime.stderr
    case('original-changed-abi-runtime-fails', exit_code=runtime.returncode)
    fresh = dependency / 'fresh-classes'
    fresh.mkdir()
    failed_compile = run([javac, '-d', str(fresh), '-cp', str(dep_jar), str(dependency / 'src/Main.java')], dependency, env)
    assert failed_compile.returncode != 0 and 'cannot find symbol' in failed_compile.stderr
    case('fresh-native-compiler-rejects-changed-abi', exit_code=failed_compile.returncode)

    testing = root / 'testing'
    generated(testing)
    write(testing / 'src/Main.java', 'public class Main { public static void main(String[] args) { } }\n')
    test_source = testing / 'src/test/ExampleTest.java'
    write(test_source, 'public class ExampleTest { @org.junit.Test public void actualFailure() { org.junit.Assert.assertEquals(1, 2); } }\n')
    (testing / 'lib').mkdir()
    shutil.copyfile(junit, testing / 'lib/junit.jar')
    original = build(testing, 'test')
    assert original.returncode == 0 and 'OK (0 tests)' in original.stdout
    case('original-failing-test-not-discovered', exit_code=0, actual_tests_run=0)
    direct = run([java, '-cp', 'build/classes:lib/*', 'org.junit.runner.JUnitCore', 'ExampleTest'], testing, env)
    assert direct.returncode != 0 and 'Tests run: 1,  Failures: 1' in direct.stdout
    case('explicit-junit4-class-failure', exit_code=direct.returncode)

    def console(classes):
        return run([java, '-jar', str(junit), 'execute', '--class-path', str(classes), '--scan-class-path', '--include-engine', 'junit-vintage', '--fail-if-no-tests', '--disable-ansi-colors', '--disable-banner', '--details', 'summary'], testing, env)

    failed = console(testing / 'build/classes')
    assert failed.returncode != 0 and '1 tests failed' in failed.stdout
    case('current-console-propagates-actual-failure', exit_code=failed.returncode)
    write(test_source, 'public class ExampleTest { @org.junit.Test public void actualSuccess() { org.junit.Assert.assertEquals(1, 1); } }\n')
    assert build(testing, 'compile').returncode == 0
    passed = console(testing / 'build/classes')
    assert passed.returncode == 0 and '1 tests successful' in passed.stdout
    case('current-console-executes-passing-test', exit_code=0)
    empty = testing / 'empty-classes'
    empty.mkdir()
    absent = console(empty)
    assert absent.returncode != 0 and '0 tests found' in absent.stdout
    case('current-console-rejects-zero-tests', exit_code=absent.returncode)
    assert len(cases) == 16
    print(json.dumps({'passed': True, 'cases': cases, 'records': records, 'sdk_release': release, 'tool_versions': versions, 'junit_version': CONTRACT['junit']['version'], 'jdk_license_sha256': hashlib.sha256((sdk / 'legal/java.base/LICENSE').read_bytes()).hexdigest(), 'model_invocations': 0, 'scope': CONTRACT['scope']}))
