"""Inventory complete non-writable dependency bytes without following symlinks."""
import hashlib
import json
import os
import stat

def inventory(root):
    rows = []
    pending = [root]
    total = 0
    while pending:
        path = pending.pop()
        info = path.lstat()
        row = {'path': path.relative_to(root).as_posix(), 'uid': info.st_uid, 'gid': info.st_gid, 'mode': stat.S_IMODE(info.st_mode)}
        assert info.st_uid == info.st_gid == 0
        if path.is_symlink():
            row.update(kind='symlink', target=os.readlink(path))
            assert path.resolve(strict=True).is_relative_to(root)
        elif path.is_dir():
            assert not row['mode'] & 0o222
            row['kind'] = 'directory'
            pending.extend(path.iterdir())
        else:
            assert path.is_file() and not row['mode'] & 0o222 and info.st_size <= 128 * 1024**2
            with path.open('rb') as stream:
                row.update(kind='file', bytes=info.st_size, sha256=hashlib.file_digest(stream, 'sha256').hexdigest())
            total += info.st_size
            assert total <= 256 * 1024**2
        rows.append(row)
        assert len(rows) <= 20000
    rows.sort(key=lambda row: row['path'])
    return rows, hashlib.sha256(json.dumps(rows, sort_keys=True, separators=(',', ':')).encode()).hexdigest()
