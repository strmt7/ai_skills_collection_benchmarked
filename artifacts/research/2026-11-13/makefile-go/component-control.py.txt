def execute(argv, cwd, env, timeout=90):
    result = subprocess.run(argv, cwd=cwd, env=env, stdin=subprocess.DEVNULL, text=True, capture_output=True, timeout=timeout)
    row = {'argv': argv, 'cwd_name': cwd.name, 'exit_code': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr}
    records.append(row)
    print(json.dumps({'native_command': row}), file=sys.stderr, flush=True)
    return result


def case(name, **values):
    cases.append({'case': name, 'expected_observation': True, **values})
    print(json.dumps({'component_case': cases[-1]}), file=sys.stderr, flush=True)


def write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding='utf-8', newline='\n')

with tempfile.TemporaryDirectory(prefix='go-make-', dir='/tmp') as temporary:
    root = Path(temporary)
    archive = Path('/tmp/qualification-sdk.tar.gz')
    def admit_sdk(expected):
        observed = hashlib.sha256(archive.read_bytes()).hexdigest()
        if observed != expected:
            raise ValueError('SDK bytes differ from the admitted digest')
        return observed
    try:
        admit_sdk('0' * 64)
    except ValueError:
        case('wrong-sdk-digest-rejected-before-execution', rejected=True)
    else:
        raise AssertionError('Wrong digest admitted')
    assert admit_sdk(CONTRACT['sdk_sha256']) == CONTRACT['sdk_sha256']
    case('copied-sdk-digest-admitted', sha256=CONTRACT['sdk_sha256'])
    sdk_parent = root / 'sdk'
    sdk_parent.mkdir()
    with tarfile.open(archive, 'r:gz') as source:
        members = source.getmembers()
        assert len(members) <= 25000 and sum(m.size for m in members) <= 512 * 1024**2
        for member in members:
            name = PurePosixPath(member.name)
            assert name.parts and name.parts[0] == 'go' and not name.is_absolute() and '..' not in name.parts

        case(kind + '-initial-no-external-dependencies', output=first.stdout, go_sum_present=False)
        original_hash = hashlib.sha256(binary.read_bytes()).hexdigest()
        write(resource, 'updated\n')
        changed = execute([*args, 'build'], project, env)
        actual = execute([str(binary)], project, env, timeout=10)
        expected = 'initial|dev' if kind == 'original' else 'updated|dev'
        assert changed.returncode == actual.returncode == 0 and actual.stdout == expected
        assert (hashlib.sha256(binary.read_bytes()).hexdigest() == original_hash) == (kind == 'original')
        case(kind + '-embedded-resource-change', output=actual.stdout, old_binary_retained=(kind == 'original'))
        if kind == 'original':
            direct = execute([str(go), 'build', '-ldflags', '-X main.version=dev', '-o', 'demo', './cmd/demo'], project, env)
            assert direct.returncode == 0 and execute([str(binary)], project, env, timeout=10).stdout == 'updated|dev'
            case('native-go-recognizes-resource-change', exit_code=0)
        resource.unlink()
        missing = execute([*args, 'build'], project, env)
        assert (missing.returncode == 0) == (kind == 'original'), records[-1]
        if kind == 'maintained':
            assert 'message.txt' in missing.stderr and 'no matching files found' in missing.stderr
        case(kind + '-missing-required-resource', exit_code=missing.returncode)
        write(resource, 'updated\n')
        passed = execute([*args, 'test'], project, env)
        assert passed.returncode == 0 and '=== RUN   TestEmbeddedMessage' in passed.stdout and '--- PASS:' in passed.stdout
        case(kind + '-actual-test-passes', exit_code=0)
        write(test, negative_test)
        failed = execute([*args, 'test'], project, env)
        assert failed.returncode != 0 and 'controlled test failure' in failed.stdout and '--- FAIL:' in failed.stdout
        case(kind + '-actual-test-failure-propagates', exit_code=failed.returncode)
    assert len(cases) == 14
