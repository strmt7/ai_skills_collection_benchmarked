def call(cwd, argv, extra_env=None):
    env = dict(os.environ)
    env.pop('TASK_SETTING', None)
    env.update(extra_env or {})
    result = subprocess.run(argv, cwd=cwd, env=env, text=True, capture_output=True, timeout=10)
    records.append({'argv': argv, 'cwd_name': cwd.name, 'exit_code': result.returncode, 'stdout': result.stdout, 'stderr': result.stderr})
    print(json.dumps({'native_command': records[-1]}), file=sys.stderr, flush=True)
    return result


def build(cwd, *args, extra_env=None):
    return call(cwd, [make, '--no-print-directory', *args], extra_env)


def write(path, text):
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(text, encoding='utf-8', newline='\n')


def case(name, **values):
    cases.append({'case': name, 'expected_observation': True, **values})
    print(json.dumps({'component_case': cases[-1]}), file=sys.stderr, flush=True)
